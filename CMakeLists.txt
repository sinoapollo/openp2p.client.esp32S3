cmake_minimum_required(VERSION 3.22)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(proxy)

# ---- config.h 出厂空值编译期提示（开源仓库默认全空；也可烧录后经 80 口网页配置台在线填）----
# 读 main/config.h，判断关键必填项是否仍为空，空则醒目提示用户先设置。
set(_cfg_h "${CMAKE_CURRENT_LIST_DIR}/main/config.h")
if(NOT EXISTS "${_cfg_h}")
  message(WARNING "未找到 main/config.h，请先准备配置模板再构建")
else()
  file(READ "${_cfg_h}" _cfg_text)
  # 宏已填（引号内非空）→ ON，否则 OFF
  function(cfg_filled _macro _outvar)
    string(REGEX MATCH "${_macro}[ \t]+\"[^\"]*\"" _hit "${_cfg_text}")
    if(NOT _hit)
      set(${_outvar} OFF PARENT_SCOPE)
      return()
    endif()
    string(REGEX REPLACE ".*\"([^\"]*)\"" "\\1" _val "${_hit}")
    if(_val STREQUAL "")
      set(${_outvar} OFF PARENT_SCOPE)
    else()
      set(${_outvar} ON PARENT_SCOPE)
    endif()
  endfunction()
  cfg_filled(OPENP2P_WIFI_SSID f_ssid)
  cfg_filled(OPENP2P_TOKEN    f_token)
  cfg_filled(OPENP2P_NODE     f_node)
  cfg_filled(OPENP2P_APP_DSTIP f_dip)
  cfg_filled(OPENP2P_DST_22    f_d22)
  cfg_filled(OPENP2P_DST_3389  f_d3389)
  cfg_filled(OPENP2P_DST_8080  f_d8080)
  cfg_filled(OPENP2P_DST_11434 f_d11434)
  set(_dst_any OFF)
  foreach(_x f_dip f_d22 f_d3389 f_d8080 f_d11434)
    if(${_x})
      set(_dst_any ON)
    endif()
  endforeach()
  set(_missing "")
  if(NOT f_ssid)
    list(APPEND _missing "WiFi SSID/密码")
  endif()
  if(NOT f_token)
    list(APPEND _missing "OPENP2P_TOKEN")
  endif()
  if(NOT f_node)
    list(APPEND _missing "OPENP2P_NODE")
  endif()
  if(NOT _dst_any)
    list(APPEND _missing "目标 IP(OPENP2P_APP_DSTIP 或 OPENP2P_DST_*)")
  endif()
  if(_missing)
    string(REPLACE ";" "、" _miss_str "${_missing}")
    message(WARNING "========================================================")
    message(WARNING "  config.h 尚未完整设置，以下必填项仍为空：${_miss_str}")
    message(WARNING "  方式一：编辑 main/config.h 填入自己的值后再编译烧录。")
    message(WARNING "  方式二：直接编译烧录，首启后访问 http://<板子IP>/ 网页配置台在线填写。")
    message(WARNING "  详见 Readme.md「首次使用」章节。")
    message(WARNING "========================================================")
  else()
    message(STATUS "config.h 已完整配置（SSID/token/节点/目标 IP 均已填）。")
  endif()
endif()
